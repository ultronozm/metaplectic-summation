% Compiles the paper with the sketches of sketches.tex; the paper is unchanged.
% Build with make sketched.
\PassOptionsToPackage{front}{sketches}
\AddToHook{begindocument/before}{\usepackage{sketches}}
\AddToHook{cmd/maketitle/after}{\printsketches}
\AddToHook{enddocument}{\input{sketches}}
% amsart sets subsections as run-in headings, which write their contents line
% only when the next paragraph starts, after a \label that follows them.  The
% outline would then list a sketch of such a subsection before its heading.
% Here a \label that directly follows a run-in heading is set with the heading.
\makeatletter
\AddToHook{begindocument/end}{%
  \let\sketched@label\label
  \DeclareRobustCommand\label[1]{%
    \ifvmode\if@noskipsec
      \g@addto@macro\@svsechd{\sketched@label{#1}}%
    \else\sketched@label{#1}\fi
    \else\sketched@label{#1}\fi}}
\makeatother
